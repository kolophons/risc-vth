#+begin_src sh
# assemble, link, and objcopy
riscv64-none-elf-as -march=rv64i -mabi=lp64 -o boot.o boot.s
NIX_BUILD_ID_STYLE=none riscv64-none-elf-ld -Ttext=0x80000000 --build-id=none -o boot.bin boot.o
riscv64-none-elf-objcopy -O binary boot.elf boot.bin
#+end_src

Run it
#+begin_src sh
qemu-system-riscv64 \
  -machine virt \
  -smp 1 -m 128M \
  -bios none -kernel boot.bin \
  -nographic
# <C-a> + x to exit.
#+end_src


With GDB:
#+begin_src sh
qemu-system-riscv64 \
  -machine virt \
  -smp 1 -m 128M \
  -bios none -kernel boot.bin \
  -nographic \
  -S -gdb tcp:127.0.0.1:1234
# <C-a> + x to exit.
#+end_src

#+begin_src sh
  gdb -q boot.elf -
#+end_src

Perform an objdump:
#+begin_src sh
# for the linked executable
riscv64-none-elf-objdump -d boot.elf

# for the object file, including the linker's outstanding work
# the additional -r shows "relocation records" alongside instructions
riscv64-none-elf-objdump -dr boot.o
  #+end_src
